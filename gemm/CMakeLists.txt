cmake_minimum_required(VERSION 3.24)

project(sm120_gemm_cute LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

find_package(CUDAToolkit REQUIRED)

# This kernel uses TMA. On Blackwell SM120, CuTe only enables the TMA
# instruction path for feature-suffixed architectures such as sm_120a.
set(CMAKE_CUDA_ARCHITECTURES 120a)

set(GEMM_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(UTILS_TOOLS_HEADER "${REPO_ROOT}/utils/tools.h")

# Generate conservative gemm_tn configurations and instantiate all of them
# for the verification target.
set(GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
set(GEMM_TN_MAX_SMEM_KB 96 CACHE STRING "Max dynamic shared memory in KiB for generated gemm_tn configs")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${GEMM_SOURCE_DIR}/gemm_configs.py"
)
execute_process(
  COMMAND python3 "${GEMM_SOURCE_DIR}/gemm_configs.py"
      --output "${GENERATED_DIR}/configs.cmake"
      --max-smem-kb "${GEMM_TN_MAX_SMEM_KB}"
  RESULT_VARIABLE GEN_RESULT
)
if(NOT GEN_RESULT EQUAL 0)
  message(FATAL_ERROR "gemm/gemm_configs.py failed with code ${GEN_RESULT}")
endif()
include("${GENERATED_DIR}/configs.cmake")

set(GEMM_TN_ONLY_CONFIG "" CACHE STRING "Only instantiate one generated gemm_tn config by name. Empty builds all configs.")
if(GEMM_TN_ONLY_CONFIG)
  set(FILTERED_GEMM_TN_CONFIGS "")
  foreach(cfg IN LISTS GEMM_TN_CONFIGS)
    string(REPLACE "|" ";" cfg_list "${cfg}")
    list(GET cfg_list 10 CFG_NAME)
    if(CFG_NAME STREQUAL GEMM_TN_ONLY_CONFIG)
      list(APPEND FILTERED_GEMM_TN_CONFIGS "${cfg}")
    endif()
  endforeach()

  if(NOT FILTERED_GEMM_TN_CONFIGS)
    message(FATAL_ERROR "GEMM_TN_ONLY_CONFIG='${GEMM_TN_ONLY_CONFIG}' was not generated by gemm_configs.py")
  endif()

  set(GEMM_TN_CONFIGS "${FILTERED_GEMM_TN_CONFIGS}")
  message(STATUS "Building only gemm_tn config: ${GEMM_TN_ONLY_CONFIG}")
endif()

set(GENERATED_KERNELS "")
set(GENERATED_CONFIG_DECLS "")
set(GENERATED_CONFIG_ENTRIES "")

foreach(cfg IN LISTS GEMM_TN_CONFIGS)
  string(REPLACE "|" ";" cfg_list "${cfg}")
  list(GET cfg_list 0 TYPE_NAME)
  list(GET cfg_list 1 TYPE_CPP)
  list(GET cfg_list 2 BM)
  list(GET cfg_list 3 BN)
  list(GET cfg_list 4 BK)
  list(GET cfg_list 5 STAGE)
  list(GET cfg_list 6 CWG)
  list(GET cfg_list 7 WARPS_M)
  list(GET cfg_list 8 WARPS_N)
  list(GET cfg_list 9 BS_W)
  list(GET cfg_list 10 CFG_NAME)

  message(STATUS "Instantiating gemm_tn config: ${CFG_NAME}")

  set(GENERATED_KERNEL "${GENERATED_DIR}/gemm_${CFG_NAME}.cu")
  configure_file("${GEMM_SOURCE_DIR}/gemm.cu.in" "${GENERATED_KERNEL}" @ONLY)
  list(APPEND GENERATED_KERNELS "${GENERATED_KERNEL}")

  string(APPEND GENERATED_CONFIG_DECLS
"extern template void gemm_tn<${TYPE_CPP}, ${BM}, ${BN}, ${BK}, ${STAGE}, ${CWG}, ${WARPS_M}, ${WARPS_N}, ${BS_W}>(
    int M,
    int N,
    int K,
    const ${TYPE_CPP}* A,
    const ${TYPE_CPP}* B,
    ${TYPE_CPP}* C,
    cudaStream_t stream);

")
  string(APPEND GENERATED_CONFIG_ENTRIES
"    {
        \"${CFG_NAME}\",
        ${BM},
        ${BN},
        ${BK},
        ${STAGE},
        ${CWG},
        ${WARPS_M},
        ${WARPS_N},
        ${BS_W},
        ${STAGE} * (${BM} * ${BK} + ${BN} * ${BK}) * 2 + ${BM} * ${BN} * 2 + 2 * ${STAGE} * static_cast<int>(sizeof(uint64_t)),
        &gemm_tn<${TYPE_CPP}, ${BM}, ${BN}, ${BK}, ${STAGE}, ${CWG}, ${WARPS_M}, ${WARPS_N}, ${BS_W}>,
    },
")
endforeach()

configure_file("${GEMM_SOURCE_DIR}/gemm_config_registry.hpp.in"
               "${GENERATED_DIR}/gemm_config_registry.hpp" @ONLY)
configure_file("${GEMM_SOURCE_DIR}/gemm_config_registry.cu.in"
               "${GENERATED_DIR}/gemm_config_registry.cu" @ONLY)

set_source_files_properties("${GEMM_SOURCE_DIR}/test.cpp" "${GENERATED_KERNELS}" "${GENERATED_DIR}/gemm_config_registry.cu" PROPERTIES LANGUAGE CUDA)

add_executable(test_gemm_cute
  "${GEMM_SOURCE_DIR}/test.cpp"
  ${GENERATED_KERNELS}
  "${GENERATED_DIR}/gemm_config_registry.cu"
  "${UTILS_TOOLS_HEADER}"
)

set_target_properties(test_gemm_cute PROPERTIES
  CUDA_ARCHITECTURES 120a
  CUDA_SEPARABLE_COMPILATION OFF
  CUDA_RESOLVE_DEVICE_SYMBOLS OFF
)

target_include_directories(test_gemm_cute PRIVATE
  ${GEMM_SOURCE_DIR}
  ${REPO_ROOT}
  ${GENERATED_DIR}
  ${REPO_ROOT}/3rd/cutlass/include
  ${REPO_ROOT}/3rd/cutlass/tools/util/include
)

target_compile_options(test_gemm_cute PRIVATE
  $<$<COMPILE_LANGUAGE:CXX>:-O3>
  $<$<COMPILE_LANGUAGE:CUDA>:-O3>
  $<$<COMPILE_LANGUAGE:CUDA>:-Xptxas=-O3>
  $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
  $<$<COMPILE_LANGUAGE:CUDA>:--expt-extended-lambda>
)

target_link_libraries(test_gemm_cute PRIVATE
  CUDA::cudart
  CUDA::cublas
)

# CMake + make build notes for test_gemm_cute.
#
# Source/configure path, relative to the repository root:
#   gemm
#
# Generated files path:
#   ${CMAKE_BINARY_DIR}/generated
#   Examples:
#     gemm/build/generated
#     gemm/build_one/generated
#
# Default full build, all generated gemm_tn configs:
#   cd gemm
#   cmake -S . -B build
#   cd build
#   make -j16 test_gemm_cute
#
# Equivalent CMake build command from the source directory:
#   cd gemm
#   cmake --build build --target test_gemm_cute -j16
#
# Build only one generated gemm_tn config:
#   cd gemm
#   cmake -S . -B build_one -DGEMM_TN_ONLY_CONFIG=bf16_bm64_bn128_bk64_s3_cwg2_wm2_wn4_bsw4
#   cd build_one
#   make -j16 test_gemm_cute
#
# Equivalent CMake build command for the one-config build:
#   cd gemm
#   cmake --build build_one --target test_gemm_cute -j16
#
# Configurable CMake cache parameters:
#   -DGEMM_TN_MAX_SMEM_KB=<KiB>
#     Max dynamic shared memory in KiB passed to gemm_configs.py.
#     Default: 96
#
#   -DGEMM_TN_ONLY_CONFIG=<config-name>
#     Instantiate only one generated gemm_tn config. Empty string builds all
#     generated configs.
#     Default: ""
#
# Fixed build settings in this file:
#   CMAKE_CXX_STANDARD=20
#   CMAKE_CUDA_STANDARD=20
#   CMAKE_CUDA_ARCHITECTURES=120a
#   test_gemm_cute CUDA_ARCHITECTURES=120a
#   CUDA_SEPARABLE_COMPILATION=OFF
#   CUDA_RESOLVE_DEVICE_SYMBOLS=OFF
#   CXX compile option: -O3
#   CUDA compile options: -O3 -Xptxas=-O3 --expt-relaxed-constexpr --expt-extended-lambda
#   Linked libraries: CUDA::cudart CUDA::cublas
#
# Generated config parameter fields:
#   bf16_bm<BM>_bn<BN>_bk<BK>_s<STAGE>_cwg<CWG>_wm<WARPS_M>_wn<WARPS_N>_bsw<BS_W>
#
# Values generated by gemm_configs.py before legality filtering:
#   type=bf16
#   BM={64,128}
#   BN={64,128}
#   BK={64}
#   STAGE={2,3}
#   CWG={2,4}
#   BS_W={4,8}
#   WARPS_M/WARPS_N from legal_warp_shapes(), with WARPS_M*WARPS_N=4*CWG.
#
# Executable paths after build:
#   gemm/build/test_gemm_cute
#   gemm/build_one/test_gemm_cute
